
// Theme toggle — CSS custom properties handle all recoloring automatically
function applyThemeVars(isLight) {
  var t = isLight ? 'light' : 'dark';
  document.body.setAttribute('data-theme', t);
  document.documentElement.setAttribute('data-theme', t);
  // Update hero overlays
  document.querySelectorAll('.hero[data-overlay-dark]').forEach(function(hero){
    var val = isLight ? hero.getAttribute('data-overlay-light') : hero.getAttribute('data-overlay-dark');
    hero.style.setProperty('--hero-overlay', val||'none');
  });
  // Update chart grid/tick colors
  var gridC = isLight ? 'rgba(26,23,20,.12)' : 'rgba(240,237,230,.06)';
  var txtC  = isLight ? '#5a5450' : '#6b6560';
  var bgC   = isLight ? '#f0ede5' : '#1a1916';
  var fgC   = isLight ? '#1a1714' : '#f0ede6';
  if(window.Chart && Chart.instances){
    Object.values(Chart.instances).forEach(function(chart){
      var sc = chart.options.scales;
      if(sc){
        ['x','y'].forEach(function(ax){
          if(sc[ax]){
            if(sc[ax].grid) sc[ax].grid.color = gridC;
            if(sc[ax].ticks) sc[ax].ticks.color = txtC;
          }
        });
      }
      var pl = chart.options.plugins;
      if(pl){
        if(pl.legend && pl.legend.labels) pl.legend.labels.color = txtC;
        if(pl.tooltip){ pl.tooltip.backgroundColor=bgC; pl.tooltip.titleColor=fgC; pl.tooltip.bodyColor=txtC; }
      }
      chart.update('none');
    });
  }
}

  // Code page (href starting "projects/"). A relative projects/... path can't
  // resolve inside the preview's blob-URL iframe, so instead of letting that
  // fail, assemble the project from the embedded copy below and open it in a
  // real new tab — done synchronously, right here in the click handler, not
  // via postMessage to the parent (a window.open() triggered from an async
  // postMessage handler loses the click's user-activation and gets silently
  // blocked by the browser — this keeps it inside the original click).
  // On a real exported/hosted site window.parent === window (no iframe), so
  // this is a no-op there and the link's own real href/target="_blank" is
  // what actually runs — the embedded copy below is preview-only weight.
  window.__customProjects = {};
  function sbAssembleCustomProjectHTML(cp) {
    var fileNames = Object.keys(cp.files);
    function findMatch(ref) {
      var clean = (ref||'').replace(/^\.?\//,'').split(/[?#]/)[0];
      for(var i=0;i<fileNames.length;i++){ if(fileNames[i]===clean || fileNames[i].slice(-(clean.length+1))==='/'+clean) return fileNames[i]; }
      return null;
    }
    var html = (cp.files[cp.entry]||{}).content || '';
    html = html.replace(/<link([^>]+)rel=["']stylesheet["']([^>]*)>/gi, function(m, pre, post){
      var hrefM = (pre+post).match(/href=["']([^"']+)["']/i);
      if(!hrefM) return m;
      var key = findMatch(hrefM[1]);
      return (key && !cp.files[key].isBinary) ? '<style>'+cp.files[key].content+'</style>' : m;
    });
    html = html.replace(/<script([^>]*)src=["']([^"']+)["']([^>]*)><\/script>/gi, function(m, pre, src, post){
      var key = findMatch(src);
      return (key && !cp.files[key].isBinary) ? '<script>'+cp.files[key].content+'<\/script>' : m;
    });
    html = html.replace(/(src|href)=["'](?!https?:|data:|\/\/|#|mailto:)([^"']+)["']/gi, function(m, attr, val){
      var key = findMatch(val);
      return (key && cp.files[key].isBinary) ? attr+'="'+cp.files[key].content+'"' : m;
    });
    return html;
  }

function fbToggleCheck(row) {
  var box = row.querySelector('.fb-check-box');
  box.classList.toggle('chk');
  var errId = row.closest('[data-fi]') ? row.closest('[data-fi]').querySelector('.fb-error-msg') : null;
  if(errId) { errId.classList.remove('show'); row.style.borderColor=''; }
}
function fbToggleMCQ(row, formId, si, fi2, minCheck, maxCheck) {
  var wrap = row.closest('[data-type="mcq"]');
  var box = row.querySelector('.fb-check-box');
  var isChecked = box.classList.contains('chk');
  if(!isChecked && maxCheck > 0) {
    var checkedNow = wrap ? wrap.querySelectorAll('.fb-mcq-item .fb-check-box.chk').length : 0;
    if(checkedNow >= maxCheck) return; // max reached
  }
  box.classList.toggle('chk');
  var err = document.getElementById(formId+'-err-'+si+'-'+fi2);
  if(err) err.classList.remove('show');
}
function fbToggleImgOpt(el, formId, si, fi2, minCheck, maxCheck) {
  var grid = document.getElementById(formId+'-imgchk-'+si+'-'+fi2);
  var isSelected = el.classList.contains('sel');
  if(!isSelected && maxCheck > 0) {
    var selCount = grid ? grid.querySelectorAll('.fb-img-opt.sel').length : 0;
    if(selCount >= maxCheck) return; // max reached
  }
  el.classList.toggle('sel');
  var err = grid ? grid.parentElement.querySelector('.fb-error-msg') : null;
  if(err) err.classList.remove('show');
}
function fbFieldChanged(inp, formId, si, fi2) {
  inp.classList.remove('fb-error');
  var err = document.getElementById(formId+'-err-'+si+'-'+fi2);
  if(err) err.classList.remove('show');
}
function fbDropdownChange(sel, formId, si, fi2) {
  sel.classList.remove('fb-error');
  var err = document.getElementById(formId+'-err-'+si+'-'+fi2);
  if(err) err.classList.remove('show');
  var customInput = document.getElementById(formId+'-ddcustom-'+si+'-'+fi2);
  if(customInput) {
    customInput.style.display = sel.value === '__custom__' ? '' : 'none';
    if(sel.value === '__custom__') customInput.focus();
  }
}
function fbValidateStep(formId, si) {
  var container = document.getElementById(formId+'-step-'+si);
  if(!container) return true;
  var valid = true;
  // text/number inputs
  container.querySelectorAll('.fb-input').forEach(function(inp) {
    var fn = inp.dataset.fn; var req = inp.dataset.required==='true';
    var err = inp.parentElement.querySelector('.fb-error-msg');
    var min = inp.min !== '' ? parseFloat(inp.min) : null;
    var max = inp.max !== '' ? parseFloat(inp.max) : null;
    var fail = false;
    if(req && !inp.value.trim()) fail = true;
    if(inp.type==='number' && inp.value!=='') {
      var v = parseFloat(inp.value);
      if(isNaN(v)) fail = true;
      if(min!==null && v<min) fail = true;
      if(max!==null && v>max) fail = true;
    }
    if(fail){ inp.classList.add('fb-error'); if(err){err.classList.add('show');} valid=false; }
    else { inp.classList.remove('fb-error'); if(err){err.classList.remove('show');} }
  });
  // selects (dropdown)
  container.querySelectorAll('.fb-select').forEach(function(sel) {
    var req = sel.dataset.required==='true';
    var wrap = sel.closest('[data-fi]');
    var err = wrap ? wrap.querySelector('.fb-error-msg') : null;
    var val = sel.value;
    // If custom is selected, validate the text input instead
    if(val === '__custom__') {
      var customInp = wrap ? wrap.querySelector('input.fb-input') : null;
      val = customInp ? customInp.value.trim() : '';
      if(req && !val) { if(err)err.classList.add('show'); sel.style.borderColor='#e05c5c'; valid=false; return; }
    }
    if(req && !val) { if(err)err.classList.add('show'); sel.style.borderColor='#e05c5c'; valid=false; }
    else { if(err)err.classList.remove('show'); sel.style.borderColor=''; }
  });
  // checkboxes — only fail if explicitly required
  container.querySelectorAll('.fb-check-row[data-required="true"]').forEach(function(row) {
    var box = row.querySelector('.fb-check-box');
    var wrap = row.closest('[data-fi]');
    var err = wrap ? wrap.querySelector('.fb-error-msg') : null;
    if(!box.classList.contains('chk')) { row.style.borderColor='#e05c5c'; if(err)err.classList.add('show'); valid=false; }
  });
  // MCQ (multi-choice checkbox) validation
  container.querySelectorAll('[data-type="mcq"]').forEach(function(wrap) {
    var req = wrap.dataset.required === 'true';
    var minCheck = parseInt(wrap.dataset.min)||0;
    var checkedCount = wrap.querySelectorAll('.fb-mcq-item .fb-check-box.chk').length;
    var errId = wrap.querySelector('.fb-error-msg');
    var fail = false;
    if(req && checkedCount === 0) fail = true;
    // If user selected something but below min, always warn
    if(checkedCount > 0 && minCheck > 0 && checkedCount < minCheck) fail = true;
    // If required and below min
    if(req && minCheck > 0 && checkedCount < minCheck) fail = true;
    if(fail) { if(errId) errId.classList.add('show'); valid = false; }
    else { if(errId) errId.classList.remove('show'); }
  });
  // image checkbox
  container.querySelectorAll('[id*="-imgchk-"]').forEach(function(grid) {
    var req = grid.dataset.required==='true';
    var minCheck = parseInt(grid.dataset.min)||0;
    var selCount = grid.querySelectorAll('.sel').length;
    var err = grid.parentElement.querySelector('.fb-error-msg');
    var fail = false;
    if(req && selCount === 0) fail = true;
    if(selCount > 0 && minCheck > 0 && selCount < minCheck) fail = true;
    if(req && minCheck > 0 && selCount < minCheck) fail = true;
    if(fail) { if(err) err.classList.add('show'); valid=false; }
    else { if(err) err.classList.remove('show'); }
  });
  return valid;
}
function fbNextStep(formId, si, total) {
  if(!fbValidateStep(formId,si)) return;
  var cur = document.getElementById(formId+'-step-'+si);
  var nxt = document.getElementById(formId+'-step-'+(si+1));
  var curNav = document.getElementById(formId+'-nav-'+si);
  var nxtNav = document.getElementById(formId+'-nav-'+(si+1));
  if(cur) cur.style.display='none';
  if(nxt) nxt.style.display='';
  if(curNav) curNav.style.display='none';
  if(nxtNav) nxtNav.style.display='';
  // progress bar
  var prog = document.getElementById(formId+'-prog');
  var lbl = document.getElementById(formId+'-step-lbl');
  if(prog) prog.style.width=((si+2)/total*100).toFixed(1)+'%';
  if(lbl) lbl.textContent='Step '+(si+2)+' of '+total;
}
function fbPrevStep(formId, si) {
  var cur = document.getElementById(formId+'-step-'+si);
  var prev = document.getElementById(formId+'-step-'+(si-1));
  var curNav = document.getElementById(formId+'-nav-'+si);

  if(cur) cur.style.display='none';
  if(prev) prev.style.display='';
  if(curNav) curNav.style.display='none';
  if(prevNav) prevNav.style.display='';
}

// Device info — auto-collected silently on page load
var _fbDeviceData = {};
function fbCollectDevice(formId) {
  // Silently collect basic device info — no button or user action needed
  _fbDeviceData[formId] = {
    _device: /Mobile|Android|iPhone|iPad/i.test(navigator.userAgent) ? 'Mobile' : 'Desktop',
    _browser: (function(){
      var ua = navigator.userAgent;
      if(ua.indexOf('Firefox')>-1) return 'Firefox';
      if(ua.indexOf('Edg')>-1) return 'Edge';
      if(ua.indexOf('Chrome')>-1) return 'Chrome';
      if(ua.indexOf('Safari')>-1) return 'Safari';
      return 'Other';
    })(),
    _os: (function(){
      var ua = navigator.userAgent;
      if(ua.indexOf('Win')>-1) return 'Windows';
      if(ua.indexOf('Mac')>-1) return 'macOS';
      if(ua.indexOf('Android')>-1) return 'Android';
      if(ua.indexOf('iPhone')>-1||ua.indexOf('iPad')>-1) return 'iOS';
      if(ua.indexOf('Linux')>-1) return 'Linux';
      return 'Unknown';
    })(),
    _screen: screen.width+'x'+screen.height,
    _timezone: Intl.DateTimeFormat().resolvedOptions().timeZone,
    _language: navigator.language||'',
    _ip:'', _lat:'', _lng:''
  };
  // Fetch IP address in background
  fetch('https://api.ipify.org?format=json').then(function(r){return r.json();}).then(function(data){
    if(_fbDeviceData[formId]) _fbDeviceData[formId]._ip = data.ip||'';
  }).catch(function(){});
  // Try geolocation (browser permission prompt — will silently fail if denied)
  if(navigator.geolocation) {
    navigator.geolocation.getCurrentPosition(function(pos){
      if(_fbDeviceData[formId]){
        _fbDeviceData[formId]._lat = pos.coords.latitude.toFixed(5);
        _fbDeviceData[formId]._lng = pos.coords.longitude.toFixed(5);
      }
    }, function(){/* denied — that's fine */});
  }
}

async function fbSubmit(formId, lastSi) {
  if(!fbValidateStep(formId, lastSi)) return;
  var section = document.getElementById(formId);
  if(!section) return;
  // Honeypot check
  var hp = section.querySelector('input[name="__hp"]');
  if(hp && hp.value) return; // bot detected — silently ignore
  var scriptUrl = section.dataset.script;
  var sheetName = section.dataset.sheet;
  var limit = parseInt(section.dataset.limit)||1;
  var redirect = section.dataset.redirect||'';
  var successMsg = section.dataset.success||'Thank you!';
  if(!scriptUrl) {
    fbShowGlobalErr(formId,'Script URL is not configured yet.');
    return;
  }
  // Submission limit check (localStorage)
  var lsKey = 'fb_sub_'+formId+'_'+window.location.pathname;
  var subCount = parseInt(localStorage.getItem(lsKey)||'0');
  if(subCount >= limit) {
    fbShowGlobalErr(formId,'You have reached the maximum number of submissions for this form.');
    return;
  }
  // Collect all field values
  var params = new URLSearchParams({ path: sheetName, action: 'write' });
  var allSteps = section.querySelectorAll('.fb-step');
  allSteps.forEach(function(step, si) {
    step.querySelectorAll('.fb-input').forEach(function(inp){ params.set(inp.dataset.fn, inp.value); });
    step.querySelectorAll('.fb-select').forEach(function(sel){
      var fn = sel.dataset.fn;
      var val = sel.value;
      // If custom entry selected, use the text input value
      if(val === '__custom__') {
        var customInput = sel.closest('[data-fi]') ? sel.closest('[data-fi]').querySelector('input.fb-input') : null;
        val = customInput ? customInput.value.trim() : '';
      }
      params.set(fn, val);
    });
    step.querySelectorAll('.fb-check-row:not(.fb-mcq-item)').forEach(function(row){
      var fn = row.dataset.fn;
      var chk = row.querySelector('.fb-check-box').classList.contains('chk');
      params.set(fn, chk ? 'true' : 'false');
    });
    // MCQ multi-checkbox
    step.querySelectorAll('[data-type="mcq"]').forEach(function(wrap){
      var fn = wrap.dataset.fn;
      var selected = [];
      wrap.querySelectorAll('.fb-mcq-item .fb-check-box.chk').forEach(function(box){
        var lbl = box.nextElementSibling;
        if(lbl) selected.push(lbl.textContent.trim());
      });
      params.set(fn, selected.join(','));
    });
    step.querySelectorAll('[id*="-imgchk-"]').forEach(function(grid){
      var fn = grid.dataset.fn;
      var selected = [];
      grid.querySelectorAll('.fb-img-opt.sel').forEach(function(opt){ selected.push(opt.dataset.val); });
      params.set(fn, selected.join(','));
    });
  });
  // Auto metadata
  params.set('_page_url', window.location.href);
  params.set('_timestamp', new Date().toISOString());
  // Device info — collected automatically if enabled (no button click needed)
  if(section.dataset.collectDevice === 'true') {
    var di = _fbDeviceData[formId]||{};
    // Always fill synchronous fields (in case auto-init missed, e.g. very fast submit)
    if(!di._device) {
      di._device = /Mobile|Android|iPhone|iPad/i.test(navigator.userAgent) ? 'Mobile' : 'Desktop';
      di._browser = (function(){ var ua=navigator.userAgent; if(ua.indexOf('Firefox')>-1)return 'Firefox'; if(ua.indexOf('Edg')>-1)return 'Edge'; if(ua.indexOf('Chrome')>-1)return 'Chrome'; if(ua.indexOf('Safari')>-1)return 'Safari'; return 'Other'; })();
      di._os = (function(){ var ua=navigator.userAgent; if(ua.indexOf('Win')>-1)return 'Windows'; if(ua.indexOf('Mac')>-1)return 'macOS'; if(ua.indexOf('Android')>-1)return 'Android'; if(ua.indexOf('iPhone')>-1||ua.indexOf('iPad')>-1)return 'iOS'; if(ua.indexOf('Linux')>-1)return 'Linux'; return 'Unknown'; })();
      di._screen = screen.width+'x'+screen.height;
      di._timezone = Intl.DateTimeFormat().resolvedOptions().timeZone;
      di._language = navigator.language||'';
    }
    ['_device','_browser','_os','_screen','_timezone','_language','_ip','_lat','_lng'].forEach(function(k){
      params.set(k, di[k]||'');
    });
  }
  // Submit
  var submitBtn = document.getElementById(formId+'-submit');
  if(submitBtn){ submitBtn.disabled=true; submitBtn.textContent='Sending…'; }
  section.classList.add('fb-submitting');
  try {
    var res = await fetch(scriptUrl+'?'+params.toString());
    var text = await res.text();
    var data = JSON.parse(text);
    if(data.status==='success') {
      localStorage.setItem(lsKey, (subCount+1).toString());
      if(redirect) { window.location.href = redirect; return; }
      var formBody = document.getElementById(formId+'-form-body');
      var successEl = document.getElementById(formId+'-success');
      var hd = section.querySelector('.fb-hd');
      var prog = section.querySelector('.fb-progress');
      var stepLbl = section.querySelector('.fb-steps-label');
      if(formBody) formBody.style.display='none';
      if(hd) hd.style.display='none';
      if(prog) prog.style.display='none';
      if(stepLbl) stepLbl.style.display='none';
      if(successEl){ successEl.style.display=''; successEl.querySelector('.fb-success-msg').textContent=successMsg; }
    } else {
      var msg = data.message || data.error || 'Submission failed. Please try again.';
      if(data.code==='DUPLICATE') msg = data.message || 'This value has already been submitted the maximum allowed times.';
      fbShowGlobalErr(formId, msg);
      if(submitBtn){ submitBtn.disabled=false; submitBtn.textContent=section.dataset.submitText||'Submit'; }
    }
  } catch(err) {
    fbShowGlobalErr(formId,'Network error. Please check your connection and try again.');
    if(submitBtn){ submitBtn.disabled=false; submitBtn.textContent='Submit'; }
  }
  section.classList.remove('fb-submitting');
}
function fbShowGlobalErr(formId, msg) {
  var el = document.getElementById(formId+'-gerr');
  if(el){ el.textContent=msg; el.classList.add('show'); setTimeout(function(){ el.classList.remove('show'); },6000); }
}
// ── Auto-collect device info on page load for all forms that have it enabled ──
(function() {
  function initDeviceInfo() {
    document.querySelectorAll('.fb-section[data-collect-device="true"]').forEach(function(section) {
      var fid = section.id;
      if(_fbDeviceData[fid] && _fbDeviceData[fid]._device) return; // already collected
      fbCollectDevice(fid);
    });
  }
  if(document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', initDeviceInfo);
  } else {
    initDeviceInfo();
  }
})();
